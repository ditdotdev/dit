# dit demo: local development on minikube
#
# Companion to dit-demo.txt (which uses the docker context). This runs
# the same lifecycle on a kubernetes context — the use case is a
# developer iterating on an app that runs in k8s and depends on a
# database schema. They add a column / table to support a new feature,
# commit the database state alongside the code change, and share both
# with the team via `dit push`. Teammates `dit pull` and `dit checkout`
# the matching commit so dev environments stay in sync.
#
# Prereqs:
#   - minikube running (driver=hyperv on Windows, driver=docker
#     elsewhere; driver=none on a CI runner). `kubectl cluster-info`
#     must succeed before starting.
#   - For `dit commit` to work: `minikube addons enable volumesnapshots
#     csi-hostpath-driver` (commits become VolumeSnapshot objects).

### install kubernetes context

./dit context install -n k8s-demo -t kubernetes

./dit context ls

### feature branch starts here — bring up postgres in k8s

./dit run postgres -n mydb -e POSTGRES_HOST_AUTH_METHOD=trust --context k8s-demo

# dit created a StatefulSet, a headless Service, and a PVC. Pod is
# named <repo>-0 by k8s convention.

kubectl get statefulset,svc,pvc,pod -l ditRepository=mydb

./dit status mydb --context k8s-demo

./dit commit mydb -m "Initial commit"

# Each commit is a CSI VolumeSnapshot — proof:

kubectl get volumesnapshot

### authenticate to dit.dev

./dit auth status

# if needed

./dit auth login --server https://dit.dev --api-key ***REMOVED***

./dit repo list

# push to remote so teammates can pull

./dit remote add https://dit.dev/robertericreeves/mydb mydb

./dit push mydb 

./dit remote log mydb

./dit repo list

### feature work: app needs a users table — add it, commit, push

kubectl exec mydb-0 -- psql -U postgres -c "CREATE TABLE users (id SERIAL PRIMARY KEY, name VARCHAR(100), email VARCHAR(100));"

./dit commit mydb -m "Added users table"

kubectl get volumesnapshot

./dit push mydb

### feature work: seed sample data, commit, push, tag the release

kubectl exec mydb-0 -- psql -U postgres -c "INSERT INTO users (name, email) VALUES ('Alice', 'alice@example.com'), ('Bob', 'bob@example.com');"

./dit commit mydb -m "Added sample users" -t v2.3.4

./dit push mydb

### checkout history — verify the schema at each commit

kubectl get volumesnapshot

./dit log mydb

# initial commit: no users table

./dit checkout mydb -c aad3f57903be45978ae402c79f80fbca

kubectl exec mydb-0 -- psql -U postgres -c "\dt"

# after CREATE TABLE: users table exists, no rows

./dit checkout mydb -c 38672f0fc7dd455ebc73a1f5c66694ee

kubectl exec mydb-0 -- psql -U postgres -c "\dt"

# after INSERT: rows are back

./dit checkout mydb -c d8b43b33634147888ce395d65677d106

kubectl exec mydb-0 -- psql -U postgres -c "SELECT * FROM users;"

### teammate flow: wipe local, clone from remote, pull specific commits

./dit stop mydb --context k8s-demo

./dit rm mydb --context k8s-demo

kubectl get statefulset,svc,pvc,pod -l ditRepository=mydb

./dit clone https://dit.dev/robertericreeves/mydb --context k8s-demo

kubectl get statefulset,svc,pvc,pod -l ditRepository=mydb

./dit log mydb

./dit remote log mydb

# pull initial commit only — bring up postgres at the empty-schema state

./dit pull mydb -c f09374af32df49e792192b2bd8046a0d

./dit checkout mydb -c f09374af32df49e792192b2bd8046a0d

# pull the schema commit, switch to it

./dit pull mydb -c 74452746c2a64535b962363e501e1976

./dit checkout mydb -c 74452746c2a64535b962363e501e1976

# checkout the tagged release directly

kubectl exec mydb-0 -- psql -U postgres -c "SELECT * FROM users;"

./dit checkout mydb -t v2.3.4

kubectl exec mydb-0 -- psql -U postgres -c "SELECT * FROM users;"

### clean up

./dit rm mydb -f --context k8s-demo

./dit repo delete robertericreeves mydb

./dit context uninstall -f k8s-demo

kubectl get statefulset,svc,pvc,pod -l ditRepository=mydb

kubectl delete volumesnapshot --all
